# The HOME menu with the background music off (music_enabled=0): its own
# sounds play, and the interface sounds still do once it is closed.

cfg log_sounds=1
cfg music_enabled=0

6500  DOWN       a sound in the main menu
8000  END
expect logorder "[Sound] menu_enter" "[Sound] move"

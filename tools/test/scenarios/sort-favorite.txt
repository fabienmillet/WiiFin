# Favourites and the library's sort & filter: Counter Film marked as a
# favourite on its page (1); in the Movies list, 2: sorted by year, only
# the favourites: Counter Film alone; then unmarked again.
server http://127.0.0.1:18096
position Counter 0
6000  A          Connect To Jellyfin
9000  A          the profile
12000 A          Movies
15000 A          Counter Film
18000 1          favourite
20000 B          back to the list
22000 2          sort & filter
22800 RIGHT      by year
23300 DOWN
23700 DOWN       show
24200 LEFT       the favourites
25000 A          apply
28000 A          the only title: Counter Film
31000 1          not a favourite any more
33000 B
35000 END
expect log [Favorite] on: HTTP 20
expect log [Library] sort Year, genre -1, show Favourites
expect log [Library] 1 titles, sorted / filtered
expect log [Favorite] off: HTTP 20
